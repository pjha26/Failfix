import { expect, it } from 'vitest';
import { sortedScores } from '../src/sorted-scores';

it('returns a separate descending array and preserves duplicates', () => {
  const scores = [3, 1, 3];
  const result = sortedScores(scores);
  expect(result).toEqual([3, 3, 1]);
  expect(result).not.toBe(scores);
  expect(scores).toEqual([3, 1, 3]);
});
